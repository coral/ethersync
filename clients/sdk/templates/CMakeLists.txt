cmake_minimum_required(VERSION 3.20)
project(TidkodSDK LANGUAGES C CXX)

add_library(tidkod_shared SHARED IMPORTED GLOBAL)
add_library(tidkod_static STATIC IMPORTED GLOBAL)
add_library(Tidkod::shared ALIAS tidkod_shared)
add_library(Tidkod::static ALIAS tidkod_static)
add_library(tidkod ALIAS tidkod_shared)
if(WIN32)
  set_target_properties(tidkod_shared PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/tidkod_bindings.dll"
    IMPORTED_IMPLIB "${CMAKE_CURRENT_LIST_DIR}/lib/tidkod_bindings.dll.lib")
  set_target_properties(tidkod_static PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/tidkod_bindings.lib")
  target_link_libraries(tidkod_static INTERFACE ws2_32 userenv bcrypt ntdll advapi32 crypt32 secur32)
elseif(APPLE)
  set_target_properties(tidkod_shared PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libtidkod_bindings.dylib")
  set_target_properties(tidkod_static PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libtidkod_bindings.a")
  target_link_libraries(tidkod_static INTERFACE "-framework Security" "-framework SystemConfiguration" "-framework CoreFoundation")
else()
  set_target_properties(tidkod_shared PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libtidkod_bindings.so")
  set_target_properties(tidkod_static PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libtidkod_bindings.a")
  target_link_libraries(tidkod_static INTERFACE pthread dl m)
endif()
foreach(linkage shared static)
  set_target_properties(tidkod_${linkage} PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/include")
endforeach()

# Consumers can add_subdirectory this SDK without building its smoke tests.
option(TIDKOD_BUILD_EXAMPLES "Build and register packaged SDK smoke tests" ON)
if(TIDKOD_BUILD_EXAMPLES)
  enable_testing()
  foreach(linkage shared static)
    foreach(sample smoke.c smoke.cpp client.c client.cpp)
      string(REPLACE "." "-" executable "tidkod-${linkage}-${sample}")
      add_executable(${executable} "examples/${sample}")
      target_link_libraries(${executable} PRIVATE tidkod_${linkage})
      target_compile_features(${executable} PRIVATE c_std_11 cxx_std_17)
      if(@NATIVE@)
        target_compile_definitions(${executable} PRIVATE TIDKOD_NATIVE)
      endif()
      if(MSVC)
        target_compile_options(${executable} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHs-c->)
      else()
        target_compile_options(${executable} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>)
      endif()
      # Copy runtime libraries beside consumers; do not rely on the build checkout.
      if(linkage STREQUAL "shared")
        add_custom_command(TARGET ${executable} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "$<TARGET_FILE:tidkod_shared>" "$<TARGET_FILE_DIR:${executable}>")
        set_target_properties(${executable} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE)
        if(APPLE)
          set_target_properties(${executable} PROPERTIES INSTALL_RPATH "@loader_path")
        elseif(UNIX)
          set_target_properties(${executable} PROPERTIES INSTALL_RPATH "$ORIGIN")
        endif()
      endif()
      add_test(NAME ${executable} COMMAND ${executable})
      set_tests_properties(${executable} PROPERTIES TIMEOUT 30)
    endforeach()
  endforeach()
endif()
