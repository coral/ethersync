cmake_minimum_required(VERSION 3.20)
project(EthersyncSDK LANGUAGES C CXX)

add_library(ethersync_shared SHARED IMPORTED GLOBAL)
add_library(ethersync_static STATIC IMPORTED GLOBAL)
add_library(Ethersync::shared ALIAS ethersync_shared)
add_library(Ethersync::static ALIAS ethersync_static)
add_library(ethersync ALIAS ethersync_shared)
if(WIN32)
  set_target_properties(ethersync_shared PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/ethersync_bindings.dll"
    IMPORTED_IMPLIB "${CMAKE_CURRENT_LIST_DIR}/lib/ethersync_bindings.dll.lib")
  set_target_properties(ethersync_static PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/ethersync_bindings.lib")
  target_link_libraries(ethersync_static INTERFACE ws2_32 userenv bcrypt ntdll advapi32 crypt32 secur32)
elseif(APPLE)
  set_target_properties(ethersync_shared PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libethersync_bindings.dylib")
  set_target_properties(ethersync_static PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libethersync_bindings.a")
  target_link_libraries(ethersync_static INTERFACE "-framework Security" "-framework SystemConfiguration" "-framework CoreFoundation")
else()
  set_target_properties(ethersync_shared PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libethersync_bindings.so")
  set_target_properties(ethersync_static PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_LIST_DIR}/lib/libethersync_bindings.a")
  target_link_libraries(ethersync_static INTERFACE pthread dl m)
endif()
foreach(linkage shared static)
  set_target_properties(ethersync_${linkage} PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/include")
endforeach()

# Consumers can add_subdirectory this SDK without building its smoke tests.
option(ETHERSYNC_BUILD_EXAMPLES "Build and register packaged SDK smoke tests" ON)
if(ETHERSYNC_BUILD_EXAMPLES)
  enable_testing()
  foreach(linkage shared static)
    foreach(sample smoke.c smoke.cpp client.c client.cpp)
      string(REPLACE "." "-" executable "ethersync-${linkage}-${sample}")
      add_executable(${executable} "examples/${sample}")
      target_link_libraries(${executable} PRIVATE ethersync_${linkage})
      target_compile_features(${executable} PRIVATE c_std_11 cxx_std_17)
      if(@NATIVE@)
        target_compile_definitions(${executable} PRIVATE ETHERSYNC_NATIVE)
      endif()
      if(MSVC)
        target_compile_options(${executable} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHs-c->)
      else()
        target_compile_options(${executable} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>)
      endif()
      # Copy runtime libraries beside consumers; do not rely on the build checkout.
      if(linkage STREQUAL "shared")
        add_custom_command(TARGET ${executable} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "$<TARGET_FILE:ethersync_shared>" "$<TARGET_FILE_DIR:${executable}>")
        set_target_properties(${executable} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE)
        if(APPLE)
          set_target_properties(${executable} PROPERTIES INSTALL_RPATH "@loader_path")
        elseif(UNIX)
          set_target_properties(${executable} PROPERTIES INSTALL_RPATH "$ORIGIN")
        endif()
      endif()
      add_test(NAME ${executable} COMMAND ${executable})
      set_tests_properties(${executable} PROPERTIES TIMEOUT 30)
    endforeach()
  endforeach()
endif()
